# Цели libFuzzer (АРХ §10). Сборка: пресет `fuzz`; smoke: scripts/ci/fuzz-smoke.sh.
function(sk_add_fuzz name)
  cmake_parse_arguments(ARG "" "" "LIBS" ${ARGN})
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE ${ARG_LIBS})
  target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_link_options(${name} PRIVATE -fsanitize=fuzzer)
  sk_set_warnings(${name})
endfunction()

sk_add_fuzz(fuzz_canon LIBS sk::canon)
sk_add_fuzz(fuzz_snapshot_reader LIBS sk::snapshot)
target_include_directories(fuzz_snapshot_reader PRIVATE ${XXHASH_INCLUDE_DIR})
target_link_libraries(fuzz_snapshot_reader PRIVATE ${XXHASH_LIBRARY})
sk_add_fuzz(fuzz_update_json LIBS sk::maxapi)
sk_add_fuzz(fuzz_init_data LIBS sk::maxapi)
sk_add_fuzz(fuzz_outbox_json LIBS sk::maxapi)
sk_add_fuzz(fuzz_demo_tsv LIBS sk::ingest_core)
sk_add_fuzz(fuzz_weighted_distance LIBS sk::verify)
sk_add_fuzz(fuzz_registry_url LIBS sk::verify)
